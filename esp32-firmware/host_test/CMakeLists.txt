cmake_minimum_required(VERSION 3.20)
project(rosserial2_host_test LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Debug)
endif()

# Strict warnings are applied to OUR targets only (see test_codec
# below). They must not leak into FetchContent dependencies (e.g.
# Catch2 has known non-virtual-dtor designs we can't fix).
set(ROSSERIAL2_STRICT_FLAGS
    -Wall -Wextra -Wpedantic -Werror
    -Wconversion -Wshadow
)

include(FetchContent)

# Catch2 v3. Pinned to a recent stable tag; no other deps required.
FetchContent_Declare(
    Catch2
    GIT_REPOSITORY https://github.com/catchorg/Catch2.git
    GIT_TAG v3.5.4
)
FetchContent_MakeAvailable(Catch2)

# The codec component is header-only. Expose it as an INTERFACE target.
add_library(rosserial2_codec INTERFACE)
target_include_directories(rosserial2_codec INTERFACE
    ${CMAKE_CURRENT_SOURCE_DIR}/../components/rosserial2_codec/include
)

# Path to the cross-language corpus (generated by the Python side).
set(CORPUS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../ros2-bridge/tests/corpus")
set(GOLDEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../ros2-bridge/tests/golden")

add_executable(test_codec
    tests/test_codec.cpp
    tests/test_resync.cpp
    tests/test_control.cpp
    tests/test_parity.cpp
    tests/test_leaf_client.cpp
)
target_link_libraries(test_codec PRIVATE rosserial2_codec Catch2::Catch2WithMain)
target_compile_options(test_codec PRIVATE ${ROSSERIAL2_STRICT_FLAGS})
target_compile_definitions(test_codec PRIVATE
    ROSSERIAL2_CORPUS_DIR="${CORPUS_DIR}"
    ROSSERIAL2_GOLDEN_DIR="${GOLDEN_DIR}"
)

include(CTest)
include(Catch)
catch_discover_tests(test_codec)
